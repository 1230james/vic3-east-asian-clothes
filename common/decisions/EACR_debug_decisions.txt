EACR_de_DEBUG_toggle_recognition = {
    is_shown = {
        has_global_variable = EACR_debug_mode
    }
    
    possible = {
        always = yes
    }
    
    when_taken = {
        if = {
            limit = {
                is_country_type = recognized
            }
            set_country_type = unrecognized
        }
        else = {
            set_country_type = recognized
        }
    }
    
    ai_chance = {
        value = 0
    }
}

EACR_de_DEBUG_become_qing = {
    is_shown = {
        has_global_variable = EACR_debug_mode
        EACR_st_is_country_tag = { TAG = CHI }
    }
    
    possible = {
        EACR_st_is_qing_china = no
    }
    
    when_taken = {
        add_primary_culture = cu:manchu
    }
    
    ai_chance = {
        value = 0
    }
}
