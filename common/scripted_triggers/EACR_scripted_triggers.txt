EACR_st_warm_season = {
    month >= 3 # month compares 0 - 11
    month <  9
}

EACR_st_is_navy = {
    OR = {
        scope:pop ?= {
            workplace ?= {
                OR = {
                    is_building_type = building_naval_administration
                    is_building_type = building_naval_fortification
                    is_building_type = building_naval_logistics_center
                }
            }
        }
        scope:character ?= {
            has_role_of_type = admiral
        }
    }
}

EACR_st_is_army = {
    OR = {
        # Soldiers & officers not working in the navy
        scope:pop ?= {
            EACR_st_is_navy = no
            OR = {
                is_pop_type = soldiers
                is_pop_type = officers
            }
        }
        
        # Generals & Armed Forces politicians
        scope:character ?= {
            OR = {
                # Generals
                has_role_of_type = general
                
                # Armed Forces IG leader who isn't a general or admiral
                AND = {
                    armed_forces_clothes_trigger = yes
                    politician_clothes_trigger = yes
                    EACR_st_is_navy = no
                }
            }
        }
    }
}

EACR_st_is_unwesternized = {
    trigger_if = {
        limit = {
            WCR_is_active_trigger = yes
        }
        WCR_TPM_is_westernized_trigger = no
    }
    trigger_else = {
        european_clothes_trigger     = no
        european_clothes_pop_trigger = no
        
        # Force military to westernize in Imperial Korea
        trigger_if = {
            limit = {
                EACR_st_owner_is_KOR_imperial_korea = yes
                OR = {
                    EACR_st_is_army = yes
                    EACR_st_is_navy = yes
                }
            }
            always = no
        }
        
        # Force Japanese rulers to westernize after Meiji Restoration is finished
        trigger_if = {
            limit = {
                EACR_st_owner_is_country_tag = { TAG = JAP }
                monarchy_clothes_trigger = yes
            }
            scope:character ?= {
                owner ?= {
                    NOT = { has_variable = japan_restoration_complete }
                }
            }
        }
    }
    
    # Treat military folks in unwesternized subjects of GBR, FRA, and RUS as westernized to make way for the subject uniforms
    trigger_if = {
        limit = {
            military_clothes_trigger = yes
            EACR_st_is_navy = yes
            EACR_st_is_army = yes
        }
        british_empire_clothes_trigger     = no
        british_empire_clothes_pop_trigger = no
        british_empire_clothes_p_trigger   = no
        french_empire_clothes_trigger      = no
        french_empire_clothes_pop_trigger  = no
        russian_empire_clothes_trigger     = no
        russian_empire_clothes_pop_trigger = no
    }
}

EACR_st_is_country_tag = {
    trigger_if = {
        limit = {
            exists = c:$TAG$
        }
        c:$TAG$ ?= THIS
    }
    trigger_else = {
        was_formed_from = $TAG$
    }
}

EACR_st_owner_is_country_tag = {
    OR = {
        scope:pop ?= {
            save_temporary_scope_as = person
        }
        scope:character ?= {
            save_temporary_scope_as = person
        }
    }
    scope:person ?= {
        owner ?= {
            EACR_st_is_country_tag = { TAG = $TAG$ }
        }
    }
}

# Will check if the character is the monarch. Will correctly handle situations where a regency is active.
EACR_st_char_is_monarch = {
    scope:character ?= {
        owner ?= { coa_monarchy_trigger = yes }
        trigger_if = {
            limit = {
                owner ?= { has_gov_regency = no }
            }
            is_ruler_of_own_country = yes
        }
        trigger_else = {
            is_heir_of_own_country = yes
        }
    }
}

# Similar to `EACR_st_char_is_monarch`. Will return false if there is a regency (since the heir is actually the monarch
# in those cases)
EACR_st_char_is_heir = {
    scope:character ?= {
        owner ?= { coa_monarchy_trigger = yes }
        trigger_if = {
            limit = {
                owner ?= { has_gov_regency = no }
            }
            is_heir_of_own_country = yes
        }
        trigger_else = {
            always = no
        }
    }
}

EACR_st_char_is_mobilized_commander = {
    scope:character ?= {
        exists = commander_military_formation
        trigger_if = {
            limit = {
                has_role_of_type = general
            }
            commander_military_formation ?= {
                is_mobilized = yes
            }
        }
        trigger_else_if = {
            limit = {
                has_role_of_type = admiral
            }
            commander_military_formation ?= {
                NOT = { current_hq ?= home_hq }
            }
        }
        trigger_else = {
            always = no # how did we get here
        }
    }
}

# Manchu queue check will now check for China's PCs instead of the ruler culture to match other "Great Qing"-related logic
EACR_st_use_chinese_manchu_queue_hairstyle = {
    EACR_st_is_unwesternized = yes
    OR = {
        scope:pop ?= {
            save_temporary_scope_as = person
        }
        scope:character ?= {
            save_temporary_scope_as = person
        }
    }
    scope:person ?= {
        trigger_if = {
            limit = {
                EACR_st_owner_is_CHI_manchu = yes
            }
            owner ?= {
                save_temporary_scope_as = home_scope
            }
        }
        trigger_else = {
            home_country ?= {
                save_temporary_scope_as = home_scope
            }
        }
    }
    OR = {
        # Manchus discriminating against Han Chinese
        AND = {
            scope:home_scope ?= {
                OR = {
                    EACR_st_owner_is_CHI_manchu  = yes
                    EACR_st_owner_is_qing_vassal = yes
                }
                NOT = {
                    has_law_or_variant = law_type:law_multicultural
                }
            }
            scope:person ?= {
                culture = {
                    has_discrimination_trait = heritage_han
                }
            }
        }
        
        # Han not discriminating against Manchus
        AND = {
            NOT = {
                scope:home_scope ?= {
                    OR = {
                        EACR_st_owner_is_CHI_han = yes
                        EACR_st_owner_is_ming_vassal = yes
                    }
                    cultural_acceptance_base = { target = cu:manchu value >= acceptance_status_4 }
                }
            }
            scope:person ?= {
                culture = cu:manchu
            }
        }
    }
}

REPLACE:chinese_manchu_queue_hairstyle_character_trigger = {

}
REPLACE:chinese_manchu_queue_hairstyle_pops_trigger = {
    exists = scope:pop
    EACR_st_use_chinese_manchu_queue_hairstyle = yes
